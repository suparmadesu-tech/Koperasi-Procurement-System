function ItemFailure({
  op,
  online,
  onRetry,
  onDismiss,
}: {
  op: ItemOpState
  online: boolean
  onRetry: () => void
  onDismiss: () => void
}) {
  const f = op.failure
  if (op.phase !== 'failed' || f === null) return null
  return (
    <div role="alert" className="space-y-2 rounded-md bg-white p-3 text-sm text-red-900 ring-1 ring-red-800/60">
      <p className="break-words">{f.message}</p>
      {f.retryable ? (
        <>
          <p>
            Belum ada konfirmasi dari server, jadi perubahan ini mungkin sudah atau belum tersimpan. Mengirim ulang aman
            dan tidak menggandakan hasil.
          </p>
          <div className="flex flex-wrap gap-2">
            <button type="button" className={btnSecondary} onClick={onRetry} disabled={!online}>
              Coba lagi
            </button>
            <button type="button" className={btnSecondary} onClick={onDismiss}>
              Hentikan percobaan
            </button>
          </div>
          <p className="text-xs">Setelah dihentikan, layar disegarkan dari server untuk melihat hasil sebenarnya.</p>
        </>
      ) : (
        <button type="button" className={btnSecondary} onClick={onDismiss}>
          Tutup
        </button>

    </div>
  )

export function EditableSessionItemCard({
  item,
  photoUrl,
  checklist,
  online,
}: {
  item: SessionItemView
  photoUrl: string | undefined
  checklist: Checklist
  online: boolean
}) {
  const { items: itemOps, bulk } = checklist
  const op = useSyncExternalStore(
    itemOps.subscribe,
    () => itemOps.getItem(item.id),
    () => itemOps.getItem(item.id),
  )
  const bulkState = useSyncExternalStore(bulk.subscribe, bulk.getState, bulk.getState)
  const [editing, setEditing] = useState<Editing>(null)
  const [text, setText] = useState('')
  const [error, setError] = useState<string | undefined>(undefined)

  const busy = op.phase === 'sending'
  const awaitingRetry = op.phase === 'failed' && op.failure !== null && op.failure.retryable
  const bulkHolds = bulkState.phase === 'confirm' || bulkState.phase === 'running' || bulkState.phase === 'halted'
  const blocked = busy || awaitingRetry || bulkHolds || !online
  const locked = busy || awaitingRetry

  function open(kind: 'quantity' | 'price') {
    setEditing(kind)
    setError(undefined)
    setText(kind === 'quantity' ? actualQuantityPrefill(item) : priceActualPrefill(item))
  }

  async function changeStatus(next: ItemStatus) {
    if (next === item.status || blocked) return
    await itemOps.run(item.id, 'session_set_status', setStatusPayload(item.id, next), statusOkText(item.name, next))
  }

  async function saveQuantity() {
    const plan = planActualQuantity(item, text)
    if (plan.kind === 'invalid') {
      setError(plan.message)
      return
    }
    setError(undefined)
    if (plan.kind === 'noop') {
      setEditing(null)
      return
    }
    const out = await itemOps.run(
      item.id,
      'session_set_quantity_actual',
      setQuantityActualPayload(item.id, plan.value),
      quantityOkText(item.name, plan.value, item.unitName),
    )
    if (out.status === 'ok') setEditing(null)
  }

  async function savePrice() {
    const plan = planPriceActual(item, text)
    if (plan.kind === 'invalid') {
      setError(plan.message)
      return
    }
    setError(undefined)
    if (plan.kind === 'noop') {
      setEditing(null)
      return
    }
    const out = await itemOps.run(
      item.id,
      'session_set_price_actual',
      setPriceActualPayload(item.id, plan.value),
      priceOkText(item.name, plan.value),
    )
    if (out.status === 'ok') setEditing(null)
  }

  async function retry() {
    const out = await itemOps.retry(item.id)
    if (out.status === 'ok') setEditing(null)
  }

  const saving = (
    <button type="submit" className={btnPrimary} disabled={locked || bulkHolds || !online}>
      {busy ? 'Menyimpan...' : 'Simpan'}
    </button>
  )

  return (
    <SessionItemCard item={item} photoUrl={photoUrl}>
      <fieldset disabled={blocked} className="min-w-0 space-y-1 pt-2">
        <legend className="text-sm font-semibold">Status {item.name}</legend>
        <div className="flex flex-wrap gap-2">
          {ITEM_STATUSES.map((s) => (
            <label
              key={s}
              className={`flex min-h-12 items-center gap-2 rounded-md bg-white px-3 text-base ring-1 ring-black/25 ${
                blocked ? 'cursor-not-allowed opacity-60' : 'cursor-pointer'
              }`}
            >
              <input
                type="radio"
                name={`status-${item.id}`}
                value={s}
                checked={item.status === s}
                onChange={() => void changeStatus(s)}
                className="h-5 w-5"
              />
              {STATUS_LABEL[s]}
            </label>
          ))}
        </div>
      </fieldset>
      {busy ? (
        <p role="status" className="text-sm">
          Menyimpan...
        </p>
      ) : null}

      {editing === null ? (
        <div className="flex flex-wrap gap-2 pt-1">
          {item.status === 'to_buy' ? (
            <p className="w-full text-xs text-ink/75">Jumlah aktual diisi setelah barang masuk keranjang.</p>

export interface FrozenStart {
  total: string | null
  unpricedCount: number | null
}

export interface ActiveView {
  startedAt: string
  itemCount: number
  groups: StatusGroup[]
  running: Totals
  frozen: FrozenStart
}

export interface CompletedView {
  endedAt: string | null
  totalActual: string | null
  unpricedPurchased: number | null
  estimate: FrozenStart
  purchasedCount: number
  returnCount: number
}

const byName = (a: SessionItemRow, b: SessionItemRow) =>
  a.product_name_snapshot.localeCompare(b.product_name_snapshot, 'id') || a.id.localeCompare(b.id)

function toItemView(i: SessionItemRow): SessionItemView {
  return {
    id: i.id,
    name: i.product_name_snapshot,
    unitName: i.unit_snapshot,
    status: i.status,
    needed: i.quantity_needed,
    actual: i.quantity_actual,
    price: effectivePriceOf(i),
    priceIsActual: i.price_actual !== null,
    subtotal: itemSubtotal(i),
    note: i.note_snapshot,
    photoPath: i.photo_path_snapshot,
    direct: i.source_item_id === null,
    allowFraction: i.allow_fraction_snapshot,
  }
}

export function buildActiveView(session: SessionRow, items: SessionItemRow[]): ActiveView {
  const live = items.filter(isLiveItem).sort(byName)
  const groups = ITEM_STATUSES.map((status) => ({
    status,
    label: STATUS_LABEL[status],
    items: live.filter((i) => i.status === status).map(toItemView),
  })).filter((g) => g.items.length > 0)

  return {
    startedAt: session.started_at,
    itemCount: live.length,
    groups,
    running: runningTotals(live),
    frozen: { total: session.total_estimate, unpricedCount: session.unpriced_items_at_start },
  }
}

// Hanya sesi completed yang punya ringkasan; sesi cancelled tidak punya layar sendiri (FR-040).
export function buildCompletedView(session: SessionRow | null, items: SessionItemRow[]): CompletedView | null {
  if (session === null || session.status !== 'completed') return null
  const live = items.filter(isLiveItem)
  return {
    endedAt: session.ended_at,
    totalActual: session.total_actual,
    unpricedPurchased: session.unpriced_items_purchased,
    estimate: { total: session.total_estimate, unpricedCount: session.unpriced_items_at_start },
    purchasedCount: statusCounts(live).purchased,
    returnCount: returnPreview(live).count,
  }
}

// Pecahan mengikuti allow_fraction_snapshot; server tetap penentu. Jumlah aktual 0 diterima (quantity_actual tidak negatif).
export function parseActualQuantityInput(text: string, opts: { allowFraction: boolean }): QuantityResult {
  const raw = text.trim().replace(',', '.')
  if (/^(0+(\.0*)?|\.0+)$/.test(raw)) return { ok: true, value: '0' }
  return parseQuantityInput(text, opts)
}

export function validateSessionCatalogForm(input: {
  productId: string
  quantityText: string
  note: string
  allowFraction: boolean
}): FormResult<{ productId: string; quantity: string; note: string | null }, 'productId' | 'quantity' | 'note'> {
  const errors: Partial<Record<'productId' | 'quantity' | 'note', string>> = {}
  if (!input.productId) errors.productId = 'Pilih barang dari katalog.'
  const note = input.note.trim()
  if (charCount(note) > LIST_TEXT_LIMITS.note) errors.note = `Catatan maksimal ${LIST_TEXT_LIMITS.note} karakter.`
  const q = parseQuantityInput(input.quantityText, { allowFraction: input.allowFraction })
  if (!q.ok) errors.quantity = q.message
  if (Object.keys(errors).length > 0 || !q.ok) return { ok: false, errors }
  return { ok: true, value: { productId: input.productId, quantity: q.value, note: note === '' ? null : note } }

export interface ItemFailure {

  code: string

  retryable: boolean

export interface ItemOpState {
  phase: 'idle' | 'sending' | 'failed'
  failure: ItemFailure | null

// Hasil terakhir yang sudah dikonfirmasi server; dibaca pembaca layar lewat satu region status.
export interface LastOk {
  seq: number
  text: string
}

export type RunOutcome = { status: 'ignored' } | { status: 'ok' } | { status: 'failed'; failure: ItemFailure }

export interface ItemOpsConfig {
  submit: SessionSubmit
  newId: () => string
  isOnline: () => boolean
  onDiscard?: () => void
}

interface AttemptRunner {
  run: (type: SessionOpType, payload: Record<string, unknown>) => Promise<ListOpResult>
  retry: () => Promise<ListOpResult | null>
  discard: () => void
  hasPending: () => boolean
}

interface Slot {
  runner: AttemptRunner
  state: ItemOpState
  okText: string
}

const IDLE: ItemOpState = { phase: 'idle', failure: null }
const OFFLINE_MESSAGE = 'Anda sedang offline. Perubahan sesi belanja memerlukan koneksi internet.'
const UNEXPECTED_MESSAGE = 'Terjadi galat tak terduga. Coba lagi.'

function kindOf(r: Extract<ListOpResult, { ok: false }>): ItemFailureKind {
  if (r.retryable) return 'network'
  return r.code === 'OFFLINE' || r.code === 'NO_CLIENT' ? 'offline' : 'rejected'
}

// Satu slot per barang: kunci double-submit per barang, op_id baru per niat baru, op_id sama untuk "Coba lagi" (SYNC-004).
// Tanpa optimistic update: nilai di layar berubah hanya setelah server menjawab dan data disegarkan (SYNC-006).
export function createItemOps(cfg: ItemOpsConfig) {
  const slots = new Map<string, Slot>()
  const listeners = new Set<() => void>()
  let last: LastOk | null = null
  let seq = 0

  const emit = () => listeners.forEach((l) => l())

  function slotOf(id: string): Slot {
    let s = slots.get(id)
    if (!s) {
      s = { runner: createAttemptRunner<SessionOpType>(cfg.submit, cfg.newId), state: IDLE, okText: '' }
      slots.set(id, s)
    }
    return s
  }

  const holdsAttempt = (s: Slot) => s.state.phase === 'failed' && s.runner.hasPending()

  function fail(s: Slot, failure: ItemFailure): RunOutcome {
    s.state = { phase: 'failed', failure }
    emit()
    return { status: 'failed', failure }
  }

  // Fase 'sending' dipasang sebelum await pertama, jadi klik ganda berikutnya langsung diabaikan.
  async function exec(s: Slot, go: () => Promise<ListOpResult | null>): Promise<RunOutcome> {
    s.state = { phase: 'sending', failure: null }
    emit()
    let r: ListOpResult | null
    try {
      r = await go()
    } catch {
      return fail(s, { kind: 'unexpected', code: 'UNEXPECTED', message: UNEXPECTED_MESSAGE, retryable: s.runner.hasPending() })
    }
    if (r === null) {
      s.state = IDLE
      emit()
      return { status: 'ignored' }
    }
    if (r.ok) {
      s.state = IDLE
      seq += 1
      last = { seq, text: s.okText }
      emit()
      return { status: 'ok' }
    }
    return fail(s, { kind: kindOf(r), code: r.code, message: r.message, retryable: r.retryable })
  }

  async function run(
    itemId: string,
    type: ItemOpType,
    payload: Record<string, unknown>,
    okText: string,
  ): Promise<RunOutcome> {
    const s = slotOf(itemId)
    if (s.state.phase === 'sending' || holdsAttempt(s)) return { status: 'ignored' }
    if (!cfg.isOnline()) {
      return fail(s, { kind: 'offline', code: 'OFFLINE', message: OFFLINE_MESSAGE, retryable: false })
    }
    s.okText = okText
    return exec(s, () => s.runner.run(type, payload))
  }

  async function retry(itemId: string): Promise<RunOutcome> {
    const s = slots.get(itemId)
    if (!s || !holdsAttempt(s)) return { status: 'ignored' }
    // Offline: jangan memanggil pengirim, supaya op_id yang menunggu tidak terbuang.
    if (!cfg.isOnline()) return { status: 'ignored' }
    return exec(s, () => s.runner.retry())
  }

  // Menutup galat penolakan, atau menghentikan percobaan yang menunggu (hasilnya tidak diketahui, jadi layar disegarkan).
  function dismiss(itemId: string): void {
    const s = slots.get(itemId)
    if (!s || s.state.phase === 'sending') return
    if (s.runner.hasPending()) {
      s.runner.discard()
      cfg.onDiscard?.()
    }
    s.state = IDLE
    emit()
  }

  function anyBusy(): boolean {
    for (const s of slots.values()) {
      if (s.state.phase === 'sending' || holdsAttempt(s)) return true
    }
    return false
  }

  return {
    getItem: (itemId: string): ItemOpState => slots.get(itemId)?.state ?? IDLE,
    getLast: (): LastOk | null => last,
    anyBusy,
    subscribe: (listener: () => void): (() => void) => {
      listeners.add(listener)
      return () => {
        listeners.delete(listener)
      }
    },
    run,
    retry,
    dismiss,
  }
}

export type ItemOps = ReturnType<typeof createItemOps>

      setBusy(true)
      setError(null)

    setError(null)

    runner.discard()
    setPending(false)
    setError(null)
    requestListRefresh()
  }, [runner])

type Submit<T extends string> = (type: T, payload: Payload, options: SubmitOptions) => Promise<ListOpResult>

  opId: string
  type: T
  payload: Payload
}

// Satu percobaan yang gagal dengan retryable disimpan, lalu dikirim ulang dengan opId dan payload yang sama.
export function createAttemptRunner<T extends string = string>(submit: Submit<T>, newId: () => string) {
  let attempt: Attempt<T> | null = null

    const r = await submit(a.type, a.payload, { opId: a.opId })
    attempt = !r.ok && r.retryable ? a : null
    return r
  }

  return {
    run: (type: T, payload: Payload): Promise<ListOpResult> => send({ opId: newId(), type, payload }),

    discard: (): void => {
      attempt = null
    },
    hasPending: (): boolean => attempt !== null,
  }
}

export type QuantityResult = { ok: true; value: string } | { ok: false; message: string }

// Koma dan titik sama-sama dianggap pemisah desimal. Pemisah ribuan tidak didukung.
export function parseQuantityInput(text: string, opts: { allowFraction: boolean }): QuantityResult {
  const raw = text.trim().replace(',', '.')
  if (raw === '') return { ok: false, message: 'Jumlah wajib diisi.' }
  if (!/^(\d+(\.\d+)?|\.\d+)$/.test(raw)) {
    return { ok: false, message: 'Jumlah harus berupa angka positif, misalnya 2 atau 1,5.' }
  }
  const d = new D(raw)
  if (d.decimalPlaces() > 3) return { ok: false, message: 'Jumlah maksimal 3 angka di belakang koma.' }
  if (d.lte(0)) return { ok: false, message: 'Jumlah harus lebih dari 0.' }
  if (d.gte(MAX_QTY)) return { ok: false, message: 'Jumlah terlalu besar.' }
  if (!opts.allowFraction && !d.isInteger()) {
    return { ok: false, message: 'Satuan ini tidak menerima jumlah berpecahan.' }
  }
  return { ok: true, value: d.toString() }
}

export interface CatalogHit {
  id: string
  name: string
  unitName: string
  allowFraction: boolean
  price: string | null
}

export interface SearchResult {
  hits: CatalogHit[]
  total: number
  activeCount: number
}

function toHit(data: CatalogData, p: CatalogData['products'][number]): CatalogHit {
  const unit = data.units.find((u) => u.id === p.unit_id)
  return { id: p.id, name: p.name, unitName: unit?.name ?? '', allowFraction: unit?.allow_fraction ?? false, price: p.price }
}

// Hanya barang aktif yang dapat ditambahkan (FR-008). Pencarian dari data lokal (FR-007).
export function searchProducts(data: CatalogData, query: string, limit: number): SearchResult {
  const q = query.trim().toLowerCase()
  const active = data.products.filter((p) => !p.is_archived)
  const matched = (q === '' ? active : active.filter((p) => p.name.toLowerCase().includes(q))).sort(
    (a, b) => a.name.localeCompare(b.name, 'id') || a.id.localeCompare(b.id),
  )
  return { hits: matched.slice(0, limit).map((p) => toHit(data, p)), total: matched.length, activeCount: active.length }
}

export function findHit(data: CatalogData, id: string): CatalogHit | null {
  const p = data.products.find((x) => x.id === id && !x.is_archived)
  return p ? toHit(data, p) : null
}

export interface CacheEntry {
  url: string
  expiresAt: number
}

// URL bertanda tangan diambil ulang bila sisa umurnya kurang dari ini.
export const REFRESH_MARGIN_MS = 5 * 60 * 1000

export function needsFetch(cache: ReadonlyMap<string, CacheEntry>, paths: string[], now: number): string[] {
  const out: string[] = []
  for (const p of new Set(paths)) {
    const e = cache.get(p)
    if (!e || e.expiresAt - now <= REFRESH_MARGIN_MS) out.push(p)
  }
  return out
}

export function snapshotUrls(
  cache: ReadonlyMap<string, CacheEntry>,
  paths: string[],
  now: number,
): Record<string, string> {
  const out: Record<string, string> = {}
  for (const p of paths) {
    const e = cache.get(p)
    if (e && e.expiresAt > now) out[p] = e.url
  }
  return out

function peek(path: string): string | null {
  const hit = cache.get(path)
  return hit && hit.expiresAt - Date.now() > REFRESH_MARGIN_MS ? hit.url : null

export function usePhotoUrl(path: string | null) {
  const [attempt, setAttempt] = useState(0)
  const [state, setState] = useState<PhotoState>({ kind: path ? 'loading' : 'none' })

  useEffect(() => {
    if (!path) {
      setState({ kind: 'none' })
      return undefined
    }
    const cached = peek(path)

        if (!cancelled) setState({ kind: 'ready', url })

      })
    return () => {
      cancelled = true
    }

    const retryOnline = () => setAttempt((n) => n + 1)

  if (!ctx) {
    decoded.release()
    return { ok: false, error: { code: 'PHOTO_NO_CANVAS', message: 'Perangkat tidak dapat memproses gambar.' } }
  }

  let mime: 'image/webp' | 'image/jpeg' = 'image/webp'
  let best: CompressedPhoto | null = null
  try {
    for (let step = firstStep(); ; ) {
      const { width, height } = dimensionsFor(decoded.width, decoded.height, step.scale)
      canvas.width = width
      canvas.height = height
      // Latar putih agar piksel transparan tidak menjadi hitam pada JPEG.
      ctx.fillStyle = '#ffffff'
      ctx.fillRect(0, 0, width, height)
      ctx.drawImage(decoded.source, 0, 0, width, height)

      let blob = await encode(canvas, mime, step.quality)
      // Browser yang tidak bisa menyandikan WebP mengembalikan PNG; beralih ke JPEG.
      if (blob && mime === 'image/webp' && blob.type !== 'image/webp') {
        mime = 'image/jpeg'
        blob = await encode(canvas, mime, step.quality)
      }
      if (!blob) {
        return { ok: false, error: { code: 'PHOTO_ENCODE', message: 'Foto gagal diproses.' } }
      }
      if (!best || blob.size < best.blob.size) best = { blob, mime }
      if (isAcceptable(blob.size)) break
      const next = nextStep(step)
      if (!next) break
      step = next
    }
  } finally {
    decoded.release()
    canvas.width = 0
    canvas.height = 0
  }

  if (!best || !isWithinLimit(best.blob.size)) {
    return {
      ok: false,
      error: {
        code: 'PHOTO_TOO_LARGE',
        message: `Foto tetap lebih besar dari ${Math.round(MAX_BYTES / 1024)} KB setelah dikompres. Pilih foto lain.`,
      },
    }
  }
  return { ok: true, data: best }
}

// Logika murni untuk foto: ukuran, urutan percobaan kompresi, path, dan pesan galat (FR-013).

export const MAX_SIDE = 1600
export const TARGET_BYTES = 200 * 1024
// Toleransi 10 persen di atas target sebelum mencoba kualitas lebih rendah.
export const ACCEPT_BYTES = Math.round(TARGET_BYTES * 1.1)
// Sama dengan file_size_limit bucket kps-photos.
export const MAX_BYTES = 614_400
export const QUALITY_STEPS = [0.85, 0.75, 0.65, 0.55, 0.45, 0.35] as const
export const MIN_SCALE = 0.4

export interface Dimensions {
  width: number
  height: number
}

export function fitWithin(width: number, height: number, max: number = MAX_SIDE): Dimensions {
  const longest = Math.max(width, height)
  const ratio = longest > max ? max / longest : 1
  return {
    width: Math.max(1, Math.round(width * ratio)),
    height: Math.max(1, Math.round(height * ratio)),
  }
}

export function dimensionsFor(width: number, height: number, scale: number): Dimensions {
  const base = fitWithin(width, height)
  return {
    width: Math.max(1, Math.round(base.width * scale)),
    height: Math.max(1, Math.round(base.height * scale)),
  }
}

export interface Step {
  quality: number
  scale: number
}

export const firstStep = (): Step => ({ quality: QUALITY_STEPS[0], scale: 1 })

const round2 = (n: number) => Math.round(n * 100) / 100

// Turunkan kualitas dulu; bila habis, kecilkan dimensi 15 persen dan ulangi dari kualitas kedua.
export function nextStep(step: Step): Step | null {
  const qualities: readonly number[] = QUALITY_STEPS
  const i = qualities.indexOf(step.quality)
  if (i >= 0 && i < qualities.length - 1) return { quality: qualities[i + 1], scale: step.scale }
  const scale = round2(step.scale * 0.85)
  return scale < MIN_SCALE ? null : { quality: qualities[1], scale }
}

export const isAcceptable = (bytes: number) => bytes <= ACCEPT_BYTES
export const isWithinLimit = (bytes: number) => bytes <= MAX_BYTES

export function extensionFor(mime: string): 'webp' | 'jpg' {
  return mime === 'image/webp' ? 'webp' : 'jpg'
}

// Path baru setiap unggahan: <workspace_id>/<uuid>.<ekstensi>, huruf kecil (FR-013).
export function newPhotoPath(
  workspaceId: string,
  mime: string,
  uuid: () => string = () => crypto.randomUUID(),
): string {
  return `${workspaceId.toLowerCase()}/${uuid().toLowerCase()}.${extensionFor(mime)}`
}

export function describeUploadError(e: { message?: string; statusCode?: string | number } | null): string {
  const text = `${e?.message ?? ''} ${e?.statusCode ?? ''}`
  if (/failed to fetch|networkerror|load failed|network request failed/i.test(text)) {
    return 'Tidak ada koneksi internet.'
  }
  if (/exceeded|too large|payload|413/i.test(text)) return 'Foto melebihi batas 600 KB.'
  if (/mime|content.?type|not supported|415/i.test(text)) return 'Jenis berkas tidak diizinkan, hanya JPEG atau WebP.'
  if (/row-level security|unauthorized|not authorized|forbidden|403|401/i.test(text)) {
    return 'Anda tidak berhak mengunggah foto, atau sesi login berakhir.'
  }
  if (/already exists|duplicate|409/i.test(text)) return 'Berkas dengan nama itu sudah ada. Coba lagi.'
  return 'Foto gagal diunggah. Coba lagi beberapa saat.'
}

import type { SupabaseClient } from '@supabase/supabase-js'
import { supabase } from '../../lib/supabase'
import { isOnline } from './catalogStore'
import { describeUploadError, newPhotoPath } from './photoPlan'
import type { Result } from './types'

const BUCKET = 'kps-photos'
const sb = supabase as unknown as SupabaseClient | null

// Selalu path baru dan tanpa upsert: berkas yang dirujuk snapshot tidak boleh tertimpa (BR-027).
export async function uploadProductPhoto(workspaceId: string, blob: Blob, mime: string): Promise<Result<string>> {
  if (!isOnline()) {
    return { ok: false, error: { code: 'OFFLINE', message: 'Mengunggah foto memerlukan koneksi internet.' } }
  }
  if (!sb) {
    return { ok: false, error: { code: 'NO_CLIENT', message: 'Supabase belum dikonfigurasi.' } }
  }
  const path = newPhotoPath(workspaceId, mime)
  try {
    const { error } = await sb.storage
      .from(BUCKET)
      .upload(path, blob, { contentType: mime, upsert: false, cacheControl: '31536000' })
    if (error) {
      return {
        ok: false,
        error: { code: 'UPLOAD_FAILED', message: describeUploadError(error as { message?: string }) },
      }
    }
    return { ok: true, data: path }
  } catch (e) {
    return {
      ok: false,
      error: { code: 'UPLOAD_FAILED', message: describeUploadError(e as { message?: string }) },
    }
  }
}
