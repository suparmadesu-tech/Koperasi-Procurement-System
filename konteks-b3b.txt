import { D, effectivePrice, parseAmount, subtotal, sumSubtotals } from '../../lib/money'
import type { ItemStatus, SessionItemRow } from './types'

export type MathItem = Pick<
  SessionItemRow,
  'status' | 'price_actual' | 'price_snapshot' | 'quantity_actual' | 'quantity_needed' | 'deleted_at' | 'source_item_id'
>

// Satu hitungan FR-050: jumlah dalam baris barang, bukan kuantitas.
export interface Totals {
  total: string
  pricedCount: number
  unpricedCount: number
  incomplete: boolean
}

type Money = ReturnType<typeof subtotal>

function safe<T>(fn: () => T): T | null {
  try {
    return fn()
  } catch {
    return null
  }
}

// Barang yang dihapus pengguna tidak dihitung di mana pun (FR-048, BR-033).
export const isLiveItem = (i: Pick<MathItem, 'deleted_at'>): boolean => i.deleted_at === null

// FR-053: price_actual bila terisi (0 sah), jika tidak price_snapshot.
const actualPrice = (i: MathItem): Money => safe(() => effectivePrice(i.price_actual, i.price_snapshot))
const snapshotPrice = (i: MathItem): Money => safe(() => parseAmount(i.price_snapshot))

export function effectivePriceOf(i: MathItem): string | null {
  const p = actualPrice(i)
  return p === null ? null : p.toFixed(0)
}

// FR-036: pembulatan(harga efektif x jumlah aktual). Null bila harga atau jumlah aktual kosong.
export function itemSubtotal(i: MathItem): string | null {
  const s = safe(() => subtotal(actualPrice(i), i.quantity_actual))
  return s === null ? null : s.toFixed(0)
}

function build(rows: Array<{ price: Money; qty: string | null }>): Totals {
  const subs = rows.map((r) => safe(() => subtotal(r.price, r.qty)))
  const unpricedCount = rows.filter((r) => r.price === null).length
  return {
    total: sumSubtotals(subs).toFixed(0),
    pricedCount: rows.length - unpricedCount,
    unpricedCount,
    incomplete: unpricedCount > 0,
  }
}

// Total berjalan: in_cart dan purchased, harga efektif x jumlah aktual. Dihitung saat ditampilkan.
export function runningTotals(items: MathItem[]): Totals {
  return build(
    items
      .filter((i) => isLiveItem(i) && (i.status === 'in_cart' || i.status === 'purchased'))
      .map((i) => ({ price: actualPrice(i), qty: i.quantity_actual })),
  )
}

// Total aktual: hanya purchased. Pratinjau sebelum sesi selesai; setelah selesai pakai nilai beku di server.
export function actualTotals(items: MathItem[]): Totals {
  return build(
    items
      .filter((i) => isLiveItem(i) && i.status === 'purchased')
      .map((i) => ({ price: actualPrice(i), qty: i.quantity_actual })),
  )
}

// Estimasi awal: price_snapshot x jumlah kebutuhan, hanya barang dari daftar.
// Hanya pemeriksaan silang; tampilan memakai nilai beku dari server (total_estimate, unpriced_items_at_start).
export function startTotals(items: MathItem[]): Totals {
  return build(
    items
      .filter((i) => isLiveItem(i) && i.source_item_id !== null)
      .map((i) => ({ price: snapshotPrice(i), qty: i.quantity_needed })),
  )
}

export function statusCounts(items: Array<Pick<MathItem, 'status' | 'deleted_at'>>): Record<ItemStatus, number> {
  const out: Record<ItemStatus, number> = { to_buy: 0, in_cart: 0, purchased: 0 }
  for (const i of items) if (isLiveItem(i)) out[i.status] += 1
  return out
}

// 3.4.1: Q_sisa = maks(0, dibutuhkan - terbeli); terbeli hanya untuk status purchased.
export function remainderOf(i: MathItem): string {
  const needed = new D(i.quantity_needed)
  const bought = i.status === 'purchased' && i.quantity_actual !== null ? new D(i.quantity_actual) : new D(0)
  const r = needed.minus(bought)
  return r.gt(0) ? r.toString() : '0'
}

// Pratinjau jumlah barang yang kembali ke daftar saat sesi diakhiri atau dibatalkan.
export function returnPreview(items: MathItem[]): { count: number } {
  return { count: items.filter((i) => isLiveItem(i) && new D(remainderOf(i)).gt(0)).length }
}

import { requestListRefresh } from '../daftar/listSync'
import { createSubmitter, type SubmitOptions } from '../daftar/submitCore'
import type { ListOpResult } from '../daftar/types'
import { isOnline } from '../katalog/catalogStore'
import { refreshSupabaseSession, sbClient } from './rpcDeps'

import { toRpcCall, type SessionOpType } from './sessionRpc'

export type SessionOpResult = ListOpResult

const core = createSubmitter({
  rpc: async (args) => {
    const call = toRpcCall(args)
    const res = await (sbClient as SupabaseClient).rpc(call.fn, call.args)
    return { data: res.data as unknown, error: res.error as unknown, status: res.status }
  },
  refreshSession: refreshSupabaseSession,
  isOnline,
  newId: () => crypto.randomUUID(),
  messageFor: sessionMessageForCode,
  offlineMessage: 'Anda sedang offline. Perubahan sesi belanja memerlukan koneksi internet.',
})

const RETURNS_TO_LIST: readonly SessionOpType[] = ['kps_start_session', 'kps_end_session', 'kps_cancel_session']
const NO_REFRESH_CODES = ['OFFLINE', 'NO_CLIENT', 'SESSION_EXPIRED']

// Titik tunggal penulisan sesi. Antarmukanya sama dengan submitListOp; Tahap 6 mengganti isinya
// menjadi antrean offline untuk operasi yang boleh offline. Gunakan opId yang sama untuk kirim ulang.
export function submitSessionOp(
  workspaceId: string,
  opType: SessionOpType,
  payload: Record<string, unknown>,
  options: SubmitOptions = {},
): Promise<SessionOpResult> {
  if (!sbClient) {
    return Promise.resolve({
      ok: false,
      opId: options.opId ?? crypto.randomUUID(),
      code: 'NO_CLIENT',
      message: 'Supabase belum dikonfigurasi.',
      retryable: false,
    })
  }
  return core(workspaceId, opType, payload, options).then((r) => {
    if (r.ok) {
      requestSessionRefresh()
      // Mulai, akhiri, dan batalkan memindahkan barang antara daftar dan sesi.
      if (RETURNS_TO_LIST.includes(opType)) requestListRefresh()
    } else if (!r.retryable && !NO_REFRESH_CODES.includes(r.code)) {
      // Ditolak server: ambil keadaan terbaru supaya layar sama dengan server.
      requestSessionRefresh()
    }
    return r
  })
}

  async (): Promise<GuardOutcome> => {

  onSettled?: (outcome: SettledOutcome) => void
}

const OFFLINE_MESSAGE = 'Anda sedang offline. Perubahan sesi belanja memerlukan koneksi internet.'
const UNEXPECTED_MESSAGE = 'Terjadi galat tak terduga. Coba lagi.'
const INITIAL: FlowState = { phase: 'idle', note: '', warning: null, failure: null, result: null }

// Pengendali alur tanpa React: kunci double-submit, kirim ulang dengan op_id sama (SYNC-004),
// dan pembedaan ditolak server dari gagal jaringan (SYNC-020). Status tidak "tersimpan" sebelum jawaban server (SYNC-006).
export function createSessionFlow(cfg: FlowConfig) {
  const runner = createAttemptRunner<SessionOpType>(cfg.submit, cfg.newId)
  let state: FlowState = INITIAL
  const listeners = new Set<() => void>()

  const set = (next: FlowState) => {
    state = next
    listeners.forEach((l) => l())
  }
  const busy = () => state.phase === 'checking' || state.phase === 'sending'

  function fail(failure: FlowFailure) {
    set({ ...state, phase: 'failed', failure, result: null })
    cfg.onSettled?.({ ok: false, failure })
  }

  const unexpected = (): FlowFailure => ({
    kind: 'unexpected',
    code: 'UNEXPECTED',
    message: UNEXPECTED_MESSAGE,
    retryable: runner.hasPending(),
  })

  function kindOf(r: Extract<ListOpResult, { ok: false }>): FlowFailureKind {
    if (r.retryable) return 'network'
    return r.code === 'OFFLINE' || r.code === 'NO_CLIENT' ? 'offline' : 'rejected'
  }

  // Fase 'sending' dipasang sebelum await pertama, jadi klik ganda berikutnya langsung ditolak.
  async function send(mode: 'run' | 'retry'): Promise<void> {
    set({ ...state, phase: 'sending', failure: null })
    try {
      if (cfg.recheckBeforeSend) {
        const g = await cfg.guard()
        if (g.kind === 'blocked') {
          fail({ kind: 'guard', code: 'BLOCKED', message: g.message, retryable: runner.hasPending() })
          return
        }
      }
      const r = mode === 'run' ? await runner.run(cfg.opType, cfg.payload()) : await runner.retry()
      if (r === null) {
        set(INITIAL)
        return
      }
      if (r.ok) {
        set({ ...state, phase: 'done', failure: null, result: r.data })
        cfg.onSettled?.({ ok: true, data: r.data })
        return
      }
      fail({ kind: kindOf(r), code: r.code, message: r.message, retryable: r.retryable })
    } catch {
      fail(unexpected())
    }
  }

  async function begin(): Promise<void> {
    const p = state.phase
    if (p === 'checking' || p === 'sending' || p === 'confirm' || p === 'confirm_warning') return
    // Masih ada percobaan menunggu: gunakan retry atau hentikan, jangan membuat op_id baru.
    if (p === 'failed' && runner.hasPending()) return
    if (!cfg.isOnline()) {
      fail({ kind: 'offline', code: 'OFFLINE', message: OFFLINE_MESSAGE, retryable: false })
      return
    }
    set({ ...INITIAL, phase: 'checking' })
    let g: GuardOutcome
    try {
      g = await cfg.guard()
    } catch {
      fail(unexpected())
      return
    }
    if (g.kind === 'blocked') {
      fail({ kind: 'guard', code: 'BLOCKED', message: g.message, retryable: false })
      return
    }
    set({ ...INITIAL, phase: 'confirm', note: g.note, warning: g.warning })
  }

  async function confirm(): Promise<void> {
    if (state.phase === 'confirm') {
      if (state.warning !== null) {
        set({ ...state, phase: 'confirm_warning' })
        return
      }
      await send('run')
      return
    }
    if (state.phase === 'confirm_warning') await send('run')
  }

  async function retry(): Promise<void> {
    if (state.phase !== 'failed' || !runner.hasPending()) return
    await send('retry')
  }

  function dismiss(): void {
    if (busy()) return
    if (runner.hasPending()) {
      runner.discard()
      cfg.onDiscard?.()
    }
    set(INITIAL)
  }

  return {
    getState: (): FlowState => state,
    subscribe: (listener: () => void): (() => void) => {
      listeners.add(listener)
      return () => {
        listeners.delete(listener)
      }
    },
    begin,
    confirm,
    retry,
    dismiss,
    hasPending: (): boolean => runner.hasPending(),
  }
}

// Satu operasi pada satu waktu per pemakai hook. Gagal retryable disimpan untuk dikirim ulang dengan op_id yang sama.
export function useListOp(workspaceId: string) {
  const runner = useMemo(
    () =>
      createAttemptRunner<ListOpType>(
        (type, payload, options) => submitListOp(workspaceId, type, payload, options),
        () => crypto.randomUUID(),
      ),
    [workspaceId],
  )
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState<string | null>(null)
  const [pending, setPending] = useState(false)
  const after = useRef<((r: OpOk) => void) | null>(null)

  const finish = useCallback(
    (r: ListOpResult | null): ListOpResult | null => {
      setBusy(false)
      setPending(runner.hasPending())
      if (r === null) return r
      setError(r.ok ? null : r.message)
      // Ditolak server: ambil data terbaru supaya layar sama dengan server.
      if (!r.ok && !r.retryable && r.code !== 'OFFLINE' && r.code !== 'NO_CLIENT') requestListRefresh()
      return r
    },
    [runner],
  )

  const run = useCallback(
    async (type: ListOpType, payload: Record<string, unknown>, onOk: (r: OpOk) => void) => {
      after.current = onOk
      setBusy(true)
      setError(null)
      const r = finish(await runner.run(type, payload))
      if (r && r.ok) onOk(r)
      return r
    },
    [runner, finish],
  )

  const retry = useCallback(async () => {
    setBusy(true)
    setError(null)
    const r = finish(await runner.retry())
    if (r && r.ok) after.current?.(r)
    return r
  }, [runner, finish])

  const discard = useCallback(() => {
    runner.discard()
    setPending(false)
    setError(null)
    requestListRefresh()
  }, [runner])

  return { busy, error, pending, run, retry, discard }

type Payload = Record<string, unknown>
type Submit<T extends string> = (type: T, payload: Payload, options: SubmitOptions) => Promise<ListOpResult>
interface Attempt<T extends string> {
  opId: string
  type: T
  payload: Payload
}

// Satu percobaan yang gagal dengan retryable disimpan, lalu dikirim ulang dengan opId dan payload yang sama.
export function createAttemptRunner<T extends string = string>(submit: Submit<T>, newId: () => string) {
  let attempt: Attempt<T> | null = null

  async function send(a: Attempt<T>): Promise<ListOpResult> {
    const r = await submit(a.type, a.payload, { opId: a.opId })
    attempt = !r.ok && r.retryable ? a : null
    return r
  }

  return {
    run: (type: T, payload: Payload): Promise<ListOpResult> => send({ opId: newId(), type, payload }),
    retry: async (): Promise<ListOpResult | null> => (attempt ? send(attempt) : null),
    discard: (): void => {
      attempt = null
    },
    hasPending: (): boolean => attempt !== null,
  }
}

        </button>
        <button type="button" className={btnPrimary} onClick={onConfirm}>
          {confirmLabel}
        </button>
      </div>
    </dialog>
  );
}
